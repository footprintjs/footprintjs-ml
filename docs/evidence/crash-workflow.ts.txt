async function artifact(path:string,format:string):Promise<ArtifactRef>{const bytes=await readFile(resolve(root,path));return artifactRef({kind:'artifact',path,format,sha256:hash(bytes),bytes:bytes.length});}

}
async function outcome(name:string,state:State,extra:any={}){
  const result={schema:'footprint-crash-resume-outcome/v1',runId:experiment.runId,phase:name,backend:state.backend,plan,
    checkpointA:state.savedA??null,checkpointB:state.savedB??null,modelA:state.modelA??null,modelB:state.modelB??null,
    restoredA:state.restoredA??null,datasets:state.partitions,receiptA:state.approvalA??null,receiptB:state.approvalB??null,
    reloadDecision:state.reloadDecision??null,staleDecision:state.staleDecision??null,freshDecision:state.freshDecision??null,
    recoveryDecision:state.recoveryDecision??null,oldHandleRejection:state.oldHandleRejection??null,optimizerProbe:state.optimizerProbe??null,
    resumeEvidence:state.resumeEvidence??null,publicationEvidence:state.publicationEvidence??null,releaseStatus:state.releaseStatus,...extra};
  await writeFile(resolve(root,name,'outcome.json'),JSON.stringify(result,null,2));return result;
}
