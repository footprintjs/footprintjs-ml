async function verifiedCopy(ref:ArtifactRef){
  const file=resolve(source,ref.path),rel=relative(source,file);
  if(rel==='..'||rel.startsWith('..'+sep)||rel.startsWith(sep))throw new Error('Source reference escaped retained run');
  const bytes=await readFile(file);if(sha(bytes)!==ref.sha256||bytes.length!==ref.bytes)throw new Error('Retained source artifact changed: '+ref.path);
  const destination=resolve(root,ref.path),relativeDestination=relative(root,destination);
  if(relativeDestination==='..'||relativeDestination.startsWith('..'+sep)||relativeDestination.startsWith(sep))throw new Error('Destination escaped new run');
  await mkdir(dirname(destination),{recursive:true});
  try{await writeFile(destination,bytes,{flag:'wx'});}catch(error:any){if(error.code!=='EEXIST')throw error;const existing=await readFile(destination);if(sha(existing)!==ref.sha256)throw new Error('Imported artifact collision');}
  return store.put(bytes,{format:ref.format});
}
async function evidenceRef(path:string,value:any,format:string){
  const ref=await putJson(value,format);await writeFile(resolve(root,path),JSON.stringify(value,null,2)+'\n',{flag:'wx'});return ref;
}

}

chart.addFunction('Publish once, lose the acknowledgment, then retry from a fresh worker',stage(async s=>{

    protocol('ml.release.retry-model-reloaded',{attempt,priorModelRef:oldModel,currentModelRef:s.modelRef,
      sameInference:oldModel.inferenceIdentity?.sha256===s.modelRef.inferenceIdentity?.sha256});
  }
  const local=createLocalModelPublisher({root:publisherRoot});
  const transport={lookup:async(request:any)=>{
    const existing=await local.lookup(request);
    if(existing)protocol('ml.release.publisher-response',{attempt,response:existing,mode:'historical-lookup',liveSessionId:trainer!.backend.sessionId});
    return existing;
  },publish:async(request:any)=>{
    const committed=await local.publish(request);
    protocol('ml.release.publisher-response',{attempt,response:committed,mode:'publish',liveSessionId:trainer!.backend.sessionId});
    if(!committed.allowed)return committed;
    if(!ackLost){
      ackLost=true;protocol('ml.release.acknowledgment-lost',{attempt,committedResponse:committed,
        note:'Actual durable publisher result exists; client acknowledgment is deliberately discarded before the FootPrint stage can commit.'});
      await trainer!.close();trainer=undefined;throw new Error('Injected acknowledgment loss after durable local commit');
    }
    return committed;
  }};

    protocol('ml.release.publication-outcome-before-retry',{attempt,result});

  s.releasedModel=plain(s.modelRef);s.releaseStatus=result.status;s.successfulRetry=plain(result);
  s.publicationAttempt={phase:'publish-and-retry',attempt,freshPublisherClient:true,liveSessionId:trainer!.backend.sessionId};
}), 'publish-with-lost-ack').retry({attempts:2,retryOn:error=>error instanceof Error&&error.message==='Injected acknowledgment loss after durable local commit'});

  const policy={...pinnedPlan.qualityPolicy,maxAgeMs};
  const shortPlan={...pinnedPlan,releaseId:'short-lived-release',idempotencyKey:'short-lived-existing',
    target:{name:'gp-expiry-existing',expectedRevision:null},qualityPolicy:policy};
  const missingPlan={...shortPlan,releaseId:'short-lived-missing',idempotencyKey:'short-lived-missing',
    target:{name:'gp-expiry-missing',expectedRevision:null}};
  const shortPin=await putJson(shortPlan,'footprint-ml/model-release-plan-json/v1');
  const missingPin=await putJson(missingPlan,'footprint-ml/model-release-plan-json/v1');
  const prep=await prepareModelRelease({planRef:shortPin,modelRef:plain(s.modelRef),store,verifyModel});
  const missingPrep=await prepareModelRelease({planRef:missingPin,modelRef:plain(s.modelRef),store,verifyModel});
  if(!prep.allowed||!missingPrep.allowed)throw new Error('Short-lived evidence expired before preparation');
  const first=await publishModelRelease({planRef:shortPin,preparedRef:prep.preparedRef!,modelRef:plain(s.modelRef),store,verifyModel,publisher:expiryPublisher});
  if(!first.allowed||first.status!=='published')throw new Error('Short-lived first publication did not complete before expiry');
  const validUntil=prep.record!.payload.validUntil!;
  await new Promise(resolveWait=>setTimeout(resolveWait,Math.max(0,validUntil-Date.now()+25)));
  const afterExpiry=Date.now();
  // Historical acknowledgement needs no live model. It does not authorize a new commit.
  const recovered=await publishModelRelease({planRef:shortPin,preparedRef:prep.preparedRef!,store,publisher:expiryPublisher});
  const missing=await publishModelRelease({planRef:missingPin,preparedRef:missingPrep.preparedRef!,store,publisher:expiryPublisher});
  const registry=await expiryPublisher.readState();
  if(afterExpiry<=validUntil||!recovered.allowed||recovered.status!=='already-published'||missing.allowed||registry.publications.length!==1||!isDeepStrictEqual(first.publisherReceipt,recovered.publisherReceipt))
    throw new Error('Expired acknowledgement was not separated from a new expired release');

    liveVerificationSupplied:false,publicationCount:registry.publications.length,registry:plain(registry),
    originalReceiptUnchanged:isDeepStrictEqual(first.publisherReceipt,recovered.publisherReceipt)};
  protocol('ml.release.expired-acknowledgement-checked',s.expiryRecovery);
}), 'recover-expired-ack');
