function start(phase,onMessage){
  const child=spawn(process.execPath,[resolve(here,'phase.mjs'),'--phase',phase,'--root',root],
    {detached:true,stdio:['ignore','pipe','pipe','ipc'],env:process.env});
  let stdout='',stderr='',failure;
  child.stdout.on('data',chunk=>stdout+=chunk);child.stderr.on('data',chunk=>stderr+=chunk);
  const stopOwnedGroup=()=>{if(child.pid>0&&child.exitCode===null&&child.signalCode===null)try{process.kill(-child.pid,'SIGKILL');}catch(error){if(error.code!=='ESRCH')throw error;}};
  child.on('message',message=>{Promise.resolve(onMessage?.(message,child)).catch(error=>{failure=error;stopOwnedGroup();});});
  const exited=new Promise((resolveExit,reject)=>{child.on('error',reject);child.on('close',(code,signal)=>resolveExit({code,signal}));});
  const finish=async(expectedSignal=null)=>{
    const timer=setTimeout(()=>{failure??=new Error('Phase exceeded its 90 second deadline.');stopOwnedGroup();},90000);
    try{const exit=await exited;await writeFile(resolve(root,phase+'-process.json'),JSON.stringify({phase,pid:child.pid,exit,stdout,stderr},null,2));
      if(failure)throw failure;if(expectedSignal)assert.equal(exit.signal,expectedSignal);else if(exit.code!==0)throw new Error(`${phase} failed: ${stderr||stdout}`);return exit;
    }finally{clearTimeout(timer);}
  };
  return {child,exited,finish,stopOwnedGroup};
}
async function waitForBarrier(proc){
  const deadline=Date.now()+65000;
  while(Date.now()<deadline){
    if(proc.child.exitCode!==null||proc.child.signalCode!==null)throw new Error('The child exited before the numerical interruption barrier.');
    try{return await json(resolve(root,'fault/barrier.json'));}catch(error){if(error.code!=='ENOENT')throw error;}
    await new Promise(resolveWait=>setTimeout(resolveWait,50));
  }
  throw new Error('No numerical interruption barrier was produced.');
}
